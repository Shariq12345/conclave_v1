\section{Formal Cryptographic Security Reduction \& Privacy Proof}
\label{sec:appendix_security_proof}

This appendix presents a formal game-based cryptographic security reduction proving that FedGuard's threshold Secure Aggregation protocol guarantees computational privacy for honest clients under the Decisional Diffie-Hellman (DDH) assumption and Shamir's information-theoretic threshold secret sharing property.

\subsection{Hardness Assumption: Decisional Diffie-Hellman (DDH)}

Let $\mathbb{G}$ be a cyclic group of prime order $q$ generated by $g \in \mathbb{G}$, and let $\lambda \in \mathbb{N}$ denote the security parameter (e.g., Curve25519 with 128-bit security level).

\begin{definition}[Decisional Diffie-Hellman Assumption]
The Decisional Diffie-Hellman (DDH) problem is computationally hard in $\mathbb{G}$ if for every probabilistic polynomial-time (PPT) adversary $\mathcal{A}_{\text{DDH}}$, the advantage:
\begin{equation}
\begin{aligned}
    \operatorname{Adv}_{\mathbb{G}}^{\text{DDH}}(\mathcal{A}_{\text{DDH}}) = \Big| &\operatorname{Pr}\big[ \mathcal{A}_{\text{DDH}}(g, g^a, g^b, g^{ab}) = 1 \big] \\
    &- \operatorname{Pr}\big[ \mathcal{A}_{\text{DDH}}(g, g^a, g^b, g^z) = 1 \big] \Big|
\end{aligned}
\end{equation}
is negligible in security parameter $\lambda$, where $a, b, z \stackrel{\$}{\leftarrow} \mathbb{Z}_q$.
\end{definition}

\subsection{Formal Security Game \texorpdfstring{$\operatorname{Exp}_{\mathcal{A}}^{\text{SecAgg}}(\lambda)$}{Exp\_SecAgg(\lambda)}}

We formalize a security game between a challenger $\mathcal{C}$ and an adversary $\mathcal{A}$ representing an honest-but-curious aggregator $\mathcal{O}$ colluding with a subset of $c < k - 1$ malicious clients $\mathcal{U}_{\text{adv}} \subset \mathcal{U}$, where $k = \lceil \frac{2}{3} N \rceil$ is the Shamir secret reconstruction threshold.

\begin{enumerate}
    \item \textbf{Setup Phase:} $\mathcal{C}$ initializes $N$ client keypairs $(\text{sk}_u, \text{pk}_u = g^{\text{sk}_u})$ for $u \in \mathcal{U}$. $\mathcal{C}$ sends all public keys $\{\text{pk}_u\}_{u \in \mathcal{U}}$ and private keys $\{\text{sk}_v\}_{v \in \mathcal{U}_{\text{adv}}}$ of compromised clients to $\mathcal{A}$.
    \item \textbf{Challenge Phase:} $\mathcal{A}$ selects two distinct candidate update vectors $x_h^{(0)}, x_h^{(1)} \in \mathbb{Z}_R^d$ for an honest target client $h \in \mathcal{U} \setminus \mathcal{U}_{\text{adv}}$.
    \item \textbf{Execution Phase:} $\mathcal{C}$ samples a random challenge bit $b \stackrel{\$}{\leftarrow} \{0, 1\}$. $\mathcal{C}$ computes the masked update vector $y_h^{(b)}$ for target client $h$:
    \begin{equation}
        y_h^{(b)} = \hat{x}_h^{(b)} + \sum_{v > h} \operatorname{PRG}(s_{h,v}) - \sum_{v < h} \operatorname{PRG}(s_{v,h}) \pmod R
    \end{equation}
    $\mathcal{C}$ transmits $y_h^{(b)}$ alongside all honest client masked updates $\{y_u\}_{u \in \mathcal{U} \setminus (\mathcal{U}_{\text{adv}} \cup \{h\})}$ to $\mathcal{A}$.
    \item \textbf{Guess Phase:} $\mathcal{A}$ outputs a guess bit $b' \in \{0, 1\}$.
\end{enumerate}

The advantage of adversary $\mathcal{A}$ in breaking Secure Aggregation privacy is defined as:
\begin{equation}
    \operatorname{Adv}_{\mathcal{A}}^{\text{SecAgg}}(\lambda) = \left| \operatorname{Pr}[b' = b] - \frac{1}{2} \right|
\end{equation}

\subsection{Main Security Theorem and Proof}

\begin{theorem}[Threshold Secure Aggregation Privacy]
\label{thm:secagg_privacy}
If X25519 Key Agreement is $(t_{\text{DDH}}, \epsilon_{\text{DDH}})$-secure under the Decisional Diffie-Hellman (DDH) assumption in $\mathbb{G}$, $\operatorname{PRG}: \{0,1\}^{256} \to \mathbb{Z}_R^d$ is a cryptographically secure pseudo-random generator, and Shamir's Secret Sharing is $(k, N)$-information-theoretically secure, then for any PPT adversary $\mathcal{A}$ corrupting at most $c < k - 1$ clients, the advantage of $\mathcal{A}$ is bounded by:
\begin{equation}
    \operatorname{Adv}_{\mathcal{A}}^{\text{SecAgg}}(\lambda) \le \binom{N}{2} \cdot \operatorname{Adv}_{\mathbb{G}}^{\text{DDH}}(\lambda) + \operatorname{Adv}_{\operatorname{PRG}}(\lambda) + \operatorname{negl}(\lambda)
\end{equation}
\end{theorem}

\noindent\textbf{Proof Sketch.} We construct a sequence of hybrid games $\mathbf{G}_0, \mathbf{G}_1, \mathbf{G}_2, \mathbf{G}_3$:

\begin{itemize}
    \item $\mathbf{G}_0$: The real execution game where pairwise shared keys $k_{u,v} = g^{\text{sk}_u \cdot \text{sk}_v}$ are generated via X25519 ECDH.
    
    \item $\mathbf{G}_1$: We modify $\mathbf{G}_0$ by replacing all ECDH shared secrets $k_{u,v}$ between honest client pairs $(u, v) \in (\mathcal{U} \setminus \mathcal{U}_{\text{adv}})^2$ with uniformly random group elements $R_{u,v} \stackrel{\$}{\leftarrow} \mathbb{G}$. By a standard hybrid argument across all $\binom{N - c}{2} \le \binom{N}{2}$ honest pairs, any PPT adversary distinguishing $\mathbf{G}_0$ from $\mathbf{G}_1$ can be used to construct a distinguisher $\mathcal{D}_{\text{DDH}}$ against the DDH problem:
    \begin{equation}
        |\operatorname{Pr}[\mathbf{G}_0 = 1] - \operatorname{Pr}[\mathbf{G}_1 = 1]| \le \binom{N}{2} \cdot \operatorname{Adv}_{\mathbb{G}}^{\text{DDH}}(\lambda)
    \end{equation}
    
    \item $\mathbf{G}_2$: We replace the pseudo-random outputs $\operatorname{PRG}(s_{u,v})$ derived from uniform seeds $s_{u,v} = \operatorname{HKDF}(R_{u,v})$ with truly uniform random noise vectors $\mathbf{u}_{u,v} \stackrel{\$}{\leftarrow} \mathbb{Z}_R^d$. By the pseudo-randomness of $\operatorname{PRG}$:
    \begin{equation}
        |\operatorname{Pr}[\mathbf{G}_1 = 1] - \operatorname{Pr}[\mathbf{G}_2 = 1]| \le \operatorname{Adv}_{\operatorname{PRG}}(\lambda)
    \end{equation}
    
    \item $\mathbf{G}_3$: We evaluate the information-theoretic security of Shamir's Secret Sharing under dropout recovery. Since adversary $\mathcal{A}$ controls $c < k - 1$ malicious nodes, $\mathcal{A}$ collects at most $c \le k - 2$ secret shares for any un-dropped honest client key $s_{h,v}$. By the Shamir threshold property, any $k - 2$ polynomial evaluations over field $\mathbb{F}_q$ reveal zero information regarding the constant term $s_{h,v}$ (the conditional entropy $H(s_{h,v} \mid \text{shares}_{\mathcal{A}}) = H(s_{h,v})$).
\end{itemize}

In Game $\mathbf{G}_3$, the masking vector $\mathbf{m}_h = \sum_{v > h} \mathbf{u}_{h,v} - \sum_{v < h} \mathbf{u}_{v,h} \pmod R$ is a sum of independent, uniformly random vectors over $\mathbb{Z}_R^d$. Consequently, $\mathbf{m}_h$ acts as a statistically perfect one-time pad over $\mathbb{Z}_R^d$. Therefore, the distribution of $y_h^{(b)} = \hat{x}_h^{(b)} + \mathbf{m}_h \pmod R$ is completely independent of the challenge bit $b$:
\begin{equation}
    \operatorname{Pr}[\mathcal{A} \text{ wins in } \mathbf{G}_3] = \frac{1}{2}
\end{equation}

Combining the hybrid probability bounds yields Theorem~\ref{thm:secagg_privacy}:
\begin{equation}
    \operatorname{Adv}_{\mathcal{A}}^{\text{SecAgg}}(\lambda) \le \binom{N}{2} \cdot \operatorname{Adv}_{\mathbb{G}}^{\text{DDH}}(\lambda) + \operatorname{Adv}_{\operatorname{PRG}}(\lambda) + \operatorname{negl}(\lambda)
\end{equation}
This completes the formal security reduction proof. \hfill $\blacksquare$
